RISCO D3 — DROP DATABASE remove o banco; SINGLE_USER WITH ROLLBACK IMMEDIATE encerra sessões e transações. Material de laboratório: confirme alvo, backup restaurável, janela e plano de retorno. Consulte COMANDOS-POR-RISCO.md na raiz.

USE [master];
GO

SELECT name, state_desc, user_access_desc
FROM sys.databases
WHERE name = N'<DATABASE_NAME>';
GO

IF DB_ID(N'<DATABASE_NAME>') IS NOT NULL
BEGIN
    ALTER DATABASE [<DATABASE_NAME>] SET SINGLE_USER WITH ROLLBACK IMMEDIATE;
    DROP DATABASE [<DATABASE_NAME>];
END;
GO
