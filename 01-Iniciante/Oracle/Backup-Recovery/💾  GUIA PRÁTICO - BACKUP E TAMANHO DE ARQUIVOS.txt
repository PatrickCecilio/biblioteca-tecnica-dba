SELECT bs.database_name,
       bs.type,
       bs.backup_start_date,
       bs.backup_finish_date,
       CAST(bs.backup_size / 1024.0 / 1024 / 1024 AS decimal(10, 2)) AS backup_size_gb,
       bmf.physical_device_name
FROM msdb.dbo.backupset AS bs
JOIN msdb.dbo.backupmediafamily AS bmf
  ON bmf.media_set_id = bs.media_set_id
WHERE bs.backup_finish_date >= DATEADD(DAY, -30, CONVERT(date, GETDATE()))
  AND bs.type = 'D'
  AND bs.is_copy_only = 0
  AND bs.is_damaged = 0
ORDER BY bs.backup_finish_date DESC;

SELECT fg.name AS filegroup_name,
       ROUND(SUM(df.size) / 128.0 / 1024.0, 3) AS total_size_gb,
       ROUND((SUM(df.size) - SUM(CONVERT(bigint, FILEPROPERTY(df.name, 'SpaceUsed')))) / 128.0 / 1024.0, 3) AS free_space_gb
FROM sys.filegroups AS fg
JOIN sys.database_files AS df
  ON df.data_space_id = fg.data_space_id
GROUP BY fg.name;
