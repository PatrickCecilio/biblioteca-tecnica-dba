-- ===============================================
-- Auditoria: eventos de ALTER SYSTEM KILL SESSION
-- (últimos 30 registros)
-- ===============================================
SELECT event_timestamp,
       dbusername,
       action_name,
       sql_text
FROM   unified_audit_trail
WHERE  action_name = 'ALTER SYSTEM'
  AND  UPPER(sql_text) LIKE 'ALTER SYSTEM KILL SESSION%'
ORDER  BY event_timestamp DESC@
FETCH FIRST 30 ROWS ONLY;

-- ===============================================
-- Sem auditoria: procurar no cache (V$SQL)
-- ===============================================
SELECT parsing_schema_name, last_active_time, sql_text
FROM   v$sql
WHERE  UPPER(sql_text) LIKE 'ALTER SYSTEM KILL SESSION%';

-- ===============================================
-- Triggers: desativar (cuidado!)
-- ===============================================
ALTER TRIGGER INFOSAUDE.EXCLUIR_SESSOES_INATIVAS DISABLE;

-- ===============================================
-- Triggers: checar status
-- ===============================================
SELECT owner, trigger_name, status
FROM   dba_triggers
WHERE  owner = 'INFOSAUDE'
AND    trigger_name = 'EXCLUIR_SESSOES_INATIVAS';

-- ===============================================
-- Profiles: verificar IDLE_TIME do perfil DEFAULT
-- (corrigido: dba_profiles)
-- ===============================================
SELECT resource_name, limit
FROM   dba_profiles
WHERE  profile = 'DEFAULT'
AND    resource_name = 'IDLE_TIME';
