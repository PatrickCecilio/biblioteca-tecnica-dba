CREATE TABLE departments (
    id NUMBER PRIMARY KEY,
    department_name VARCHAR2(100) NOT NULL
);

CREATE TABLE employees (
    id NUMBER PRIMARY KEY,
    name VARCHAR2(100) NOT NULL,
    salary NUMBER(10,2) NOT NULL,
    department_id NUMBER,
      FOREIGN KEY (department_id)
      REFERENCES departments(id)
);

CREATE TABLE projects (
    id NUMBER PRIMARY KEY,
    project_name VARCHAR2(100) NOT NULL
);

CREATE TABLE employee_projects (
    employee_id NUMBER,
    project_id NUMBER,
    PRIMARY KEY (employee_id, project_id),
      FOREIGN KEY (employee_id)
      REFERENCES employees(id),
      FOREIGN KEY (project_id)
      REFERENCES projects(id)
);

CREATE TABLE salary_logs (
    log_id NUMBER PRIMARY KEY,
    employee_id NUMBER,
    old_salary NUMBER(10,2),
    new_salary NUMBER(10,2),
    change_date DATE DEFAULT SYSDATE,
      FOREIGN KEY (employee_id)
      REFERENCES employees(id)
);

CREATE SEQUENCE seq_departments START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE seq_employees START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE seq_projects START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE TRIGGER trg_salary_log
BEGIN
    INSERT INTO salary_logs (log_id, employee_id, old_salary, new_salary, change_date)
    VALUES (seq_salary_logs.NEXTVAL, :OLD.id, :OLD.salary, :NEW.salary, SYSDATE);
/

CREATE OR REPLACE VIEW high_salary_employees AS
SELECT * FROM employees
WHERE salary > 7000;
/

SELECT e.id, e.name, e.salary, d.department_name
FROM employees e
INNER JOIN departments d ON e.department_id = d.id
WHERE d.id = 2;

SELECT e.id, e.name, e.salary
FROM employees e
WHERE e.salary = (SELECT MAX(salary) FROM employees);

SELECT e.id, e.name, COUNT(ep.project_id) AS total_projects
FROM employees e
LEFT JOIN employee_projects ep ON e.id = ep.employee_id
GROUP BY e.id, e.name
ORDER BY total_projects DESC;
