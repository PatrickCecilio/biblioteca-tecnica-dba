SELECT owner, trigger_name, triggering_event, status
FROM dba_triggers
WHERE triggering_event LIKE '%LOGON%' OR triggering_event LIKE '%LOGOFF%' OR triggering_event LIKE '%LOGON, LOGOFF%';

SELECT trigger_name, trigger_type, triggering_event, description, trigger_body
FROM dba_triggers
WHERE trigger_body LIKE '%KILL%' OR trigger_body LIKE '%SESSION%';
