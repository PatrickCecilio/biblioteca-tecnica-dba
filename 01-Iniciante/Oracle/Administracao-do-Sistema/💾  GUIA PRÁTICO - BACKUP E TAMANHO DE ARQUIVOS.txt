SELECT bs.database_name, bs.type, bs.backup_start_date, bs.backup_finish_date,
CAST(bs.backup_size / 1024.0 / 1024 / 1024 AS DECIMAL(10,2)) AS backup_size_gb,
bmf.physical_device_name
FROM msdb.dbo.backupset bs
JOIN msdb.dbo.backupmediafamily bmf ON bs.media_set_id = bmf.media_set_id
WHERE bs.backup_finish_date >= DATEADD(DAY, -30, CAST(GETDATE() AS DATE))
AND bs.type = 'D'
AND bs.is_copy_only = 0
AND bs.is_damaged = 0
ORDER BY bs.backup_finish_date DESC;


#Espaço ocupado por Filegroups (SQL Server)
SELECT fg.name AS FileGroupName,
ROUND((SUM(df.size) / 128.0) / 1024.0, 3) AS TotalSizeGB,
ROUND(((SUM(df.size) - SUM(CAST(FILEPROPERTY(df.name, 'SpaceUsed') AS INT))) / 128.0) / 1024.0, 5) AS FreeSpaceGB
FROM sys.filegroups AS fg
JOIN sys.database_files AS df ON fg.data_space_id = df.data_space_id
GROUP BY fg.name;
