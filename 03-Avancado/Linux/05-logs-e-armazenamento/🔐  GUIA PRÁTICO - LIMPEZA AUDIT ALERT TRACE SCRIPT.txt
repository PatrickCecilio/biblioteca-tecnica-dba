#!/usr/bin/env bash
# Limpeza de diagnósticos Oracle — D3 (pré-visualização)
# Runbook: ../../Oracle/Performance-Monitoramento/RUNBOOK-LIMPEZA-DIAGNOSTICOS.md
# Este exemplo LISTA candidatos; não remove arquivos. Não copie caminhos de outros ambientes.
# Laboratório fictício: TARGET=/u01/app/oracle/admin/ORCL/adump RETENTION_DAYS=30
# Saída SIMULADA (não comprova execução):
# Alvo: /u01/app/oracle/admin/ORCL/adump
# Retenção: 30 dias
# /u01/app/oracle/admin/ORCL/adump/audit001.aud
# /u01/app/oracle/admin/ORCL/adump/audit002.aud
# /u01/app/oracle/admin/ORCL/adump/audit003.aud
# Defina TARGET e RETENTION_DAYS com os valores validados antes de executar.

set -euo pipefail
: "${TARGET:?Defina o diretório validado de audit ou trace}"
: "${RETENTION_DAYS:?Defina a retenção aprovada em dias}"
case "$RETENTION_DAYS" in ''|*[!0-9]*) echo 'Retenção inválida' >&2; exit 1;; esac
[ "$RETENTION_DAYS" -ge 2 ] || { echo 'Retenção menor que 2 dias exige revisão específica' >&2; exit 1; }
[ -d "$TARGET" ] && [ ! -L "$TARGET" ] || { echo 'Alvo inválido' >&2; exit 1; }
RESOLVED=$(realpath -- "$TARGET")
printf 'Alvo: %s\nRetenção: %s dias\n' "$RESOLVED" "$RETENTION_DAYS"
find "$RESOLVED" -xdev -type f -name '*.aud' -mtime +"$RETENTION_DAYS" -print
# A execução de remoção é uma etapa separada, após aprovação e inspeção da listagem.
