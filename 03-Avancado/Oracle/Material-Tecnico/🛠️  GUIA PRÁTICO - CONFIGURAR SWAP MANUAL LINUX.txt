# Ver tamanho atual
free -m
swapon -s

# Criar arquivo swap de 11 GB
dd if=/dev/zero of=/root/meuswap bs=1024 count=$((11*1024*1024))

# Definir permissões e ativar
chmod 600 /root/meuswap
chown root:root /root/meuswap
mkswap /root/meuswap
swapon /root/meuswap

# Tornar swap permanente
echo "/root/meuswap swap swap defaults 0 0" >> /etc/fstab
systemctl daemon-reexec
