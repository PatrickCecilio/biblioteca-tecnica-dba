Pré-requisitos do SO:

dnf install -y oracle-database-preinstall-19c

CodeReady Builder e algumas libs usadas com 19c em OL8:

dnf config-manager --set-enabled ol8_codeready_builder || true
dnf install -y compat-openssl11 libnsl libnsl2 libnsl2-devel ksh \
               unixODBC unixODBC-devel smartmontools net-tools

Crie a árvore de diretórios e acerte permissões:
mkdir -p /u01/app/oracle/product/19.3.0/dbhome_1
mkdir -p /u01/app/oraInventory
mkdir -p /u01/reco
mkdir -p /u01/oradata
chown -R oracle:oinstall /u01
chmod -R 775 /u01

Preparar variáveis de ambiente:
su - oracle

export ORACLE_BASE=/u01/app/oracle
export ORACLE_HOME=/u01/app/oracle/product/19.3.0/dbhome_1
export ORA_INVENTORY=/u01/app/oraInventory
export PATH=$ORACLE_HOME/bin:$PATH
export LD_LIBRARY_PATH=$ORACLE_HOME/lib:/lib:/usr/lib

Descompactar o db_home no ORACLE_HOME:
cd $ORACLE_HOME
unzip -oq LINUX.X64_193000_db_home.zip

Rodar o instalador do Database
export CV_ASSUME_DISTID=OEL7.6

cd $ORACLE_HOME

./runInstaller -silent \
 -responseFile $ORACLE_HOME/install/response/db_install.rsp \
 oracle.install.option=INSTALL_DB_SWONLY \
 UNIX_GROUP_NAME=oinstall \
 INVENTORY_LOCATION=/u01/app/oraInventory \
 ORACLE_BASE=/u01/app/oracle \
 ORACLE_HOME=/u01/app/oracle/product/19.3.0/dbhome_1 \
 oracle.install.db.InstallEdition=SE2 \
 oracle.install.db.OSDBA_GROUP=dba \
 oracle.install.db.OSOPER_GROUP=oper \
 oracle.install.db.OSBACKUPDBA_GROUP=backupdba \
 oracle.install.db.OSDGDBA_GROUP=dgdba \
 oracle.install.db.OSKMDBA_GROUP=kmdba \
 oracle.install.db.OSRACDBA_GROUP=racdba \
 DECLINE_SECURITY_UPDATES=true

Executar os scripts pós-instalação:
/u01/app/oraInventory/orainstRoot.sh
/u01/app/oracle/product/19.3.0/dbhome_1/root.sh

. oraenv
CDB1

instalando em modo silente:
==================================================================

dbca -silent -createDatabase \
  -templateName General_Purpose.dbc \
  -gdbname cdbprd1 \
  -sid cdbprd1 \
  -characterSet WE8ISO8859P1 \
  -createAsContainerDatabase true \
  -numberOfPDBs 1 \
  -pdbName ORASX01 \
  -datafileDestination "/u01/oradata/" \
  -recoveryAreaDestination "/u01/reco/" \
  -recoveryAreaSize 20480 \
  -emConfiguration NONE \
  -sysPassword 'system' \
  -systemPassword 'system' \
  -dbsnmpPassword 'system' \
  -pdbAdminPassword 'system' \
  -sampleSchema false \
  -databaseType MULTIPURPOSE \
  -totalMemory 2048
