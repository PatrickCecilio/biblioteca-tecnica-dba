sqlplus <USERNAME>/<PASSWORD>@<SERVICE_NAME> as sysdba
ALTER SESSION SET CONTAINER = XEPDB1;

SET LINESIZE 200
COL OCCUPANT_NAME FORMAT A30
SELECT OCCUPANT_NAME, SPACE_USAGE_KBYTES
FROM V$SYSAUX_OCCUPANTS
ORDER BY SPACE_USAGE_KBYTES DESC;

COL SEGMENT_NAME FORMAT A30
COL OWNER FORMAT A10
COL SEGMENT_TYPE FORMAT A15
SELECT * FROM (
SELECT SEGMENT_NAME, OWNER, SEGMENT_TYPE,
BYTES/1024/1024 AS SIZE_MB
FROM   DBA_SEGMENTS
WHERE  TABLESPACE_NAME='SYSAUX'
ORDER  BY BYTES DESC
) WHERE ROWNUM <= 10;

SELECT TABLESPACE_NAME,
ROUND(SUM(BYTES)/1024/1024)     AS SIZE_MB,
ROUND(SUM(MAXBYTES)/1024/1024)  AS MAXSIZE_MB
FROM DBA_DATA_FILES
GROUP BY TABLESPACE_NAME;

SELECT TASK_NAME, PARAMETER_NAME, PARAMETER_VALUE
FROM DBA_ADVISOR_PARAMETERS
WHERE TASK_NAME='AUTO_STATS_ADVISOR_TASK'
AND PARAMETER_NAME='EXECUTION_DAYS_TO_EXPIRE';

BEGIN
DBMS_ADVISOR.SET_TASK_PARAMETER(
task_name => 'AUTO_STATS_ADVISOR_TASK',
parameter => 'EXECUTION_DAYS_TO_EXPIRE',
value     => 1);
END;
/

EXEC PRVT_ADVISOR.DELETE_EXPIRED_TASKS;

ALTER TABLE WRI$_ADV_OBJECTS MOVE PARALLEL 2;

ALTER INDEX WRI$_ADV_OBJECTS_PK     REBUILD;
ALTER INDEX WRI$_ADV_OBJECTS_IDX_01 REBUILD;
ALTER INDEX WRI$_ADV_OBJECTS_IDX_02 REBUILD;

SELECT INDEX_NAME, STATUS
FROM DBA_INDEXES
WHERE TABLE_NAME='WRI$_ADV_OBJECTS';

SELECT DBMS_STATS.GET_STATS_HISTORY_RETENTION FROM DUAL;

SELECT DBMS_STATS.GET_STATS_HISTORY_AVAILABILITY FROM DUAL;

EXEC DBMS_STATS.ALTER_STATS_HISTORY_RETENTION(10);

EXEC DBMS_STATS.PURGE_STATS(SYSDATE - 10);

ALTER DATABASE DATAFILE '<path_do_sysaux.dbf>' RESIZE <novo_tamanho>M;

SELECT
(SELECT ROUND(SUM(BYTES)/1024/1024)
FROM DBA_SEGMENTS WHERE TABLESPACE_NAME='SYSAUX') AS USADO_MB,
(SELECT ROUND(SUM(BYTES)/1024/1024)
FROM DBA_FREE_SPACE WHERE TABLESPACE_NAME='SYSAUX') AS LIVRE_MB,
(SELECT ROUND(SUM(MAXBYTES)/1024/1024)
FROM DBA_DATA_FILES WHERE TABLESPACE_NAME='SYSAUX') AS TAMANHO_MAX_MB
FROM DUAL;

SELECT OCCUPANT_NAME, SPACE_USAGE_KBYTES
FROM V$SYSAUX_OCCUPANTS
WHERE SPACE_USAGE_KBYTES > 0
ORDER BY SPACE_USAGE_KBYTES DESC;

SELECT TASK_NAME, STATUS, LAST_EXECUTION
FROM DBA_ADVISOR_TASKS
WHERE TASK_NAME='AUTO_STATS_ADVISOR_TASK';

EXEC DBMS_STATS.GATHER_SCHEMA_STATS('SYS');

CREATE TABLE TESTE_SPACIO (ID NUMBER) TABLESPACE USERS;
