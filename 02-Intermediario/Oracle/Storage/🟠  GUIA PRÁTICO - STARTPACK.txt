SELECT tablespace_name, status, contents
FROM dba_tablespaces
WHERE tablespace_name = 'PERFSTAT';

SELECT snap_id, snap_level, TO_CHAR(snap_time, 'DD/MM/YYYY HH24:MI:SS') AS snap_time
FROM stats$snapshot
ORDER BY snap_id DESC;
