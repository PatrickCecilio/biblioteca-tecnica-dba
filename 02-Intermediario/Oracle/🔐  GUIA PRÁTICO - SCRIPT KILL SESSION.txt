CREATE TABLE session_kill_log (
    log_id         NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    sid            NUMBER,
    serial#        NUMBER,
    username       VARCHAR2(30),
    action         VARCHAR2(100),
    log_timestamp  TIMESTAMP DEFAULT SYSTIMESTAMP
);
-------------------------------------------------------------------------------------------
DECLARE
    v_blocking_count INTEGER;

    -- Variável para controlar o tempo de execução do script (em minutos)
    v_max_runtime INTEGER := 30; -- Defina o tempo máximo de execução (30 minutos, por exemplo)
    v_start_time TIMESTAMP := SYSTIMESTAMP;

BEGIN
    -- Loop principal
    LOOP
        -- Verificar se o tempo máximo de execução foi atingido
        IF (SYSTIMESTAMP - v_start_time) > (v_max_runtime / 1440) THEN
            INSERT INTO session_kill_log (action, log_timestamp)
            VALUES ('Tempo máximo de execução atingido. Encerrando o script.', SYSTIMESTAMP);
            COMMIT;
            EXIT;
        END IF;

        -- Buscar sessões bloqueadoras com mais de 1 minuto de espera
        FOR rec IN (
            SELECT blocking_session, sid, serial#, wait_class, seconds_in_wait, username
            FROM v$session
            WHERE blocking_session IS NOT NULL
            AND seconds_in_wait > 60  -- Mais de 1 minuto
            AND username NOT IN ('SYS', 'SYSTEM') -- Ignorar sessões do sistema
            ORDER BY blocking_session
        ) LOOP
            -- Contar o número de sessões bloqueadoras
            SELECT COUNT(*) INTO v_blocking_count
            FROM v$session
            WHERE blocking_session IS NOT NULL
            AND username NOT IN ('SYS', 'SYSTEM'); -- Ignorar sessões do sistema

            -- Se houver mais de 1 sessão bloqueadora, matar as sessões
            IF v_blocking_count > 1 THEN
                -- Gerar comando para matar a sessão bloqueadora
                EXECUTE IMMEDIATE 'ALTER SYSTEM KILL SESSION ''' || rec.sid || ',' || rec.serial# || ''' IMMEDIATE';

                -- Inserir log na tabela session_kill_log
                INSERT INTO session_kill_log (sid, serial#, username, action)
                VALUES (rec.sid, rec.serial#, rec.username, 'Sessão bloqueadora foi encerrada.');
                COMMIT;
            END IF;
        END LOOP;

        -- Espera 10 segundos antes de verificar novamente
        DBMS_SESSION.sleep(10);

    END LOOP;
END;
/
