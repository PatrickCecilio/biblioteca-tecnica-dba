# RISCO D3: material de laboratório. Identifique DB_UNIQUE_NAME, política de
# retenção, destino e backups recuperáveis antes de qualquer DELETE.
# Conectar ao RMAN
target /;

# Backup full com compressão\	BACKUP AS COMPRESSED BACKUPSET DATABASE;
# Backup full com TAG
BACKUP DATABASE TAG='bkp_full';
# Backup incremental nível 0 em seções (paralelismo)
BACKUP INCREMENTAL LEVEL=0 SECTION SIZE 500M DATABASE;
# Backup de tablespace específica
BACKUP TABLESPACE USERS;
# Backup com criptografia
SET ENCRYPTION IDENTIFIED BY "<SENHA_SUBSTITUIR>" ONLY;
BACKUP DATABASE;

# Configurar canais e políticas\CONFIGURE CHANNEL DEVICE TYPE DISK FORMAT '/u01/backup/%U.bkp';
CONFIGURE RETENTION POLICY TO REDUNDANCY 1;
CONFIGURE CHANNEL DEVICE TYPE DISK MAXPIECESIZE 2G;

# Inspeção: CROSSCHECK atualiza o repositório RMAN; não remove arquivos físicos.
# EXPIRED significa ausente no destino consultado, não "antigo".
SHOW RETENTION POLICY;
SHOW ARCHIVELOG DELETION POLICY;
LIST BACKUP SUMMARY;
REPORT OBSOLETE;
CROSSCHECK BACKUP;
CROSSCHECK ARCHIVELOG ALL;
REPORT OBSOLETE;

# Etapa D3 separada, somente após conferir a saída, dependências de
# recuperação/standby e aprovação. Mantenha o prompt interativo.
DELETE OBSOLETE;
