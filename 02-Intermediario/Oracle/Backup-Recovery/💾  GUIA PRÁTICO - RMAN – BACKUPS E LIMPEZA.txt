# Conectar ao RMAN
target /;

# Backup full com compressão\	BACKUP AS COMPRESSED BACKUPSET DATABASE;
# Backup full com TAG
BACKUP DATABASE TAG='bkp_full';
# Backup incremental nível 0 em seções (paralelismo)
BACKUP INCREMENTAL LEVEL=0 SECTION SIZE 500M DATABASE;
# Backup de tablespace específica
BACKUP TABLESPACE USERS;
# Backup com criptografia
SET ENCRYPTION IDENTIFIED BY "minhasenha" ONLY;
BACKUP DATABASE;

# Configurar canais e políticas\CONFIGURE CHANNEL DEVICE TYPE DISK FORMAT '/u01/backup/%U.bkp';
CONFIGURE RETENTION POLICY TO REDUNDANCY 1;
CONFIGURE CHANNEL DEVICE TYPE DISK MAXPIECESIZE 2G;

# Limpeza de backups
CROSSCHECK BACKUP;
DELETE EXPIRED BACKUP;
DELETE NOPROMPT OBSOLETE;
CROSSCHECK BACKUP;
CROSSCHECK ARCHIVELOG ALL;
DELETE NOPROMPT EXPIRED BACKUP;
DELETE NOPROMPT EXPIRED ARCHIVELOG ALL;
