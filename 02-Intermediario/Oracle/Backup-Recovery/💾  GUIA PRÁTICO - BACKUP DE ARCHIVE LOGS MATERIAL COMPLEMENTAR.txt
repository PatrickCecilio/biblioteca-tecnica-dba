# RISCO D3: validar banco, backup, destino e Data Guard antes de excluir logs.
SHOW ARCHIVELOG DELETION POLICY;
LIST ARCHIVELOG ALL;
BACKUP ARCHIVELOG ALL;

Arquivar o redo atual e fazer backup sem exclusão automática:
RUN {
  SQL 'ALTER SYSTEM ARCHIVE LOG CURRENT';
  BACKUP AS COMPRESSED BACKUPSET FORMAT '<BACKUP_PATH_VALIDADO>' ARCHIVELOG ALL;
  CROSSCHECK ARCHIVELOG ALL;
  REPORT OBSOLETE;
}

# Após conferir backups, política de retenção e dependências, uma etapa
# operacional separada pode usar DELETE OBSOLETE com prompt e aprovação.
