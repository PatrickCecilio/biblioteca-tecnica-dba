sqlplus / as sysdba

ALTER SESSION SET CONTAINER=CDB$ROOT;

SELECT name, open_mode
FROM v$pdbs;

ALTER PLUGGABLE DATABASE <SOURCE_PDB> OPEN READ ONLY;

CREATE PLUGGABLE DATABASE <TARGET_PDB>
FROM <SOURCE_PDB>
FILE_NAME_CONVERT = ('<SOURCE_PDB_PATH>', '<TARGET_PDB_PATH>');

ALTER PLUGGABLE DATABASE <TARGET_PDB> OPEN;

ALTER PLUGGABLE DATABASE <TARGET_PDB> SAVE STATE;

SELECT name, open_mode
FROM v$pdbs;
