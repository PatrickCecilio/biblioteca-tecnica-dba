RISCO D3 — antes de remover a PDB de origem, confirme CDB, nome da PDB,
backup recuperável, arquivos, serviços e validação funcional do destino.
Este exemplo é de laboratório; não copie caminhos nem execute DROP por sequência.

CREATE PLUGGABLE DATABASE PDB1_MOVED
  USING '/u01/pdb_migration/PDB1/pdb1.xml'
  FILE_NAME_CONVERT = (
    '/u01/oradata/CDB1/pdb1/', '/u01/oradata/CDB2'
  );

ALTER PLUGGABLE DATABASE <PDB_NAME> OPEN UPGRADE;
ALTER SESSION SET CONTAINER = PDB1_MOVED;
ALTER PLUGGABLE DATABASE <PDB_NAME> CLOSE IMMEDIATE;
ALTER PLUGGABLE DATABASE <PDB_NAME> OPEN;
