Não remova a PDB de origem como parte automática do fluxo de clonagem.

CREATE PLUGGABLE DATABASE PDB1_MOVED
  USING '/u01/pdb_migration/PDB1/pdb1.xml'
  FILE_NAME_CONVERT = (
    '/u01/oradata/CDB1/pdb1/', '/u01/oradata/CDB2'
  );

Se aparecer erro pedindo upgrade, faça:

ALTER PLUGGABLE DATABASE PDB1_MOVED OPEN UPGRADE;
ALTER SESSION SET CONTAINER = PDB1_MOVED;
