@echo off
setlocal

:: Altere aqui para a pasta raiz dos seus .txt
:: RISCO D3: valide o diretório; não use arquivos sensíveis neste exemplo.
set "ROOT=<DIRETORIO_VALIDADO>"

:: Arquivo de saída
set "TARGET=%ROOT%\unificado.txt"

:: Evite sobrescrever uma saída anterior.
if exist "%TARGET%" (
  echo O arquivo de saida ja existe. Escolha outro nome.
  exit /b 1
)

:: Loop recursivo (usa %%F porque está dentro de um .bat)
for /R "%ROOT%" %%F in (*.txt) do (
  if /I not "%%~fF"=="%TARGET%" (
    echo ==== %%~nxF ====>>"%TARGET%"
    type "%%~fF">>"%TARGET%"
    echo.>>"%TARGET%"
  )
)

echo.
echo Concatenação concluída em:
echo   %TARGET%
echo.
endlocal
