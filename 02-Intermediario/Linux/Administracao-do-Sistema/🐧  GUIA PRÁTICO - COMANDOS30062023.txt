-- 2. Monitoramento de Sessões Inativas (por máquina)
--------------------------------------------------------------------------------
SELECT
    MACHINE,
    COUNT(*) AS QTD_SESSOES_INATIVAS
FROM
    V$SESSION
WHERE
    STATUS = 'INACTIVE'
GROUP BY
    MACHINE
ORDER BY
    QTD_SESSOES_INATIVAS DESC;

--------------------------------------------------------------------------------
-- 3. Detalhamento de Sessões Inativas de uma Máquina Específica
--------------------------------------------------------------------------------
-- Substitua 'FMTHVD\SALUX237921' pela máquina desejada
SELECT
    s.sid,
    s.serial#,
    s.status,
    s.username,
    s.osuser,
    s.program,
    s.machine,
    s.module,
    s.action,
    s.logon_time,
    s.last_call_et,
    q.sql_text
FROM
    v$session s
LEFT JOIN
    v$sql q
  ON
    s.sql_id = q.sql_id
WHERE
    s.status  = 'INACTIVE'
  AND
    s.machine = 'FMTHVD\SALUX237921';

-- 5. Verificação de Objetos Inválidos
--------------------------------------------------------------------------------
-- Listar objetos inválidos no esquema UDBD03
SELECT
    object_type,
    object_name
FROM
    dba_objects
WHERE
    owner  = 'INFOSAUDE'
  AND
    status = 'INVALID';

-- Contagem de objetos inválidos (status 4=INVALID, 5=ERROR, 6=UNUSABLE)
SELECT
    COUNT(*) AS invalid_objs
FROM
    obj$
WHERE
    status IN (4, 5, 6);

-- 7. Status de Jobs Scheduler UTL_RECOMP_SLAVE
--------------------------------------------------------------------------------
-- Todos os jobs agendados

    job_name,
    state
FROM
    dba_scheduler_jobs

    job_name LIKE 'UTL_RECOMP_SLAVE_%';

-- Jobs atualmente em execução
